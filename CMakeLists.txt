cmake_minimum_required(VERSION 3.24)
project(CyRSAssistant VERSION 0.7.1 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CyRSAssistant requires Windows x64")
endif()
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/third_party/reshade/include/reshade.hpp")
  message(FATAL_ERROR "Run python tools/bootstrap.py first")
endif()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
file(GLOB CYRS_LOCALES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/companion/locales/*.json")
set(CYRS_LOCALE_HEADER "${CMAKE_BINARY_DIR}/generated/localization_data.hpp")
add_custom_command(OUTPUT "${CYRS_LOCALE_HEADER}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/generate_locales.py" "${CYRS_LOCALE_HEADER}"
  DEPENDS ${CYRS_LOCALES} tools/generate_locales.py companion/localization.py
  VERBATIM)
add_custom_target(cyrs_locales DEPENDS "${CYRS_LOCALE_HEADER}")
configure_file(src/resources.rc.in "${CMAKE_BINARY_DIR}/generated/resources.rc" @ONLY)
add_library(CyRSAssistant SHARED "${CMAKE_BINARY_DIR}/generated/resources.rc" src/branding.cpp src/addon.cpp src/bridge.cpp src/shader_tools.cpp src/dxc_tools.cpp src/pipeline_snapshot.cpp src/shader_lab.cpp)
add_dependencies(CyRSAssistant cyrs_locales)
target_compile_features(CyRSAssistant PRIVATE cxx_std_17)
target_include_directories(CyRSAssistant PRIVATE third_party/reshade/include third_party/imgui third_party/json src "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(CyRSAssistant PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX ImTextureID=ImU64)
target_compile_options(CyRSAssistant PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(CyRSAssistant PRIVATE advapi32 d3dcompiler dxguid bcrypt)
target_link_options(CyRSAssistant PRIVATE "/PDBALTPATH:CyRSAssistant.pdb")
set_target_properties(CyRSAssistant PROPERTIES PREFIX "" SUFFIX ".addon64" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
