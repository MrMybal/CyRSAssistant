cmake_minimum_required(VERSION 3.24)
project(CyRSAssistant VERSION 0.10.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CyRSAssistant requires Windows x64")
endif()
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/third_party/reshade/include/reshade.hpp")
  message(FATAL_ERROR "Run python tools/bootstrap.py first")
endif()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
file(GLOB CYRS_LOCALES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/companion/locales/*.json")
set(CYRS_LOCALE_HEADER "${CMAKE_BINARY_DIR}/generated/localization_data.hpp")
add_custom_command(OUTPUT "${CYRS_LOCALE_HEADER}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/generate_locales.py" "${CYRS_LOCALE_HEADER}"
  DEPENDS ${CYRS_LOCALES} tools/generate_locales.py companion/localization.py
  VERBATIM)
add_custom_target(cyrs_locales DEPENDS "${CYRS_LOCALE_HEADER}")
file(GLOB_RECURSE CYRS_HELP_INPUTS CONFIGURE_DEPENDS
  "${CMAKE_SOURCE_DIR}/.CyDocBuild/manual/*.md" "${CMAKE_SOURCE_DIR}/.CyDocBuild/assets/*")
set(CYRS_MANUAL_HEADER "${CMAKE_BINARY_DIR}/generated/manual_data.hpp")
add_custom_command(OUTPUT "${CYRS_MANUAL_HEADER}" "${CMAKE_SOURCE_DIR}/Documentation/Documentation.html"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/.CyDocBuild/build.py" --header "${CYRS_MANUAL_HEADER}"
  DEPENDS ${CYRS_HELP_INPUTS} CMakeLists.txt companion/transport.py
    .CyDocBuild/cydoc.config.json .CyDocBuild/cydoc.lock.json .CyDocBuild/package.json .CyDocBuild/package-lock.json
    .CyDocBuild/build.py .CyDocBuild/export.mjs .CyDocBuild/check.mjs .CyDocBuild/api-revisions.json
    .CyDocBuild/reader.mjs .CyDocBuild/redirect.mjs .CyDocBuild/reader.html .CyDocBuild/reader.css
    .CyDocBuild/vendor/cydoc-0.1.0.tgz
  VERBATIM)
add_custom_target(cyrs_manual DEPENDS "${CYRS_MANUAL_HEADER}" "${CMAKE_SOURCE_DIR}/Documentation/Documentation.html")
configure_file(src/resources.rc.in "${CMAKE_BINARY_DIR}/generated/resources.rc" @ONLY)
add_library(CyRSAssistant SHARED "${CMAKE_BINARY_DIR}/generated/resources.rc" src/branding.cpp src/manual.cpp src/manual_model.cpp src/addon.cpp src/bridge.cpp src/shader_tools.cpp src/dxc_tools.cpp src/pipeline_snapshot.cpp src/shader_lab.cpp)
add_dependencies(CyRSAssistant cyrs_locales cyrs_manual)
target_compile_features(CyRSAssistant PRIVATE cxx_std_17)
target_include_directories(CyRSAssistant PRIVATE third_party/reshade/include third_party/imgui third_party/json src "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(CyRSAssistant PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX ImTextureID=ImU64)
target_compile_options(CyRSAssistant PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(CyRSAssistant PRIVATE advapi32 d3dcompiler dxguid bcrypt shell32 shlwapi)
target_link_options(CyRSAssistant PRIVATE "/PDBALTPATH:CyRSAssistant.pdb")
set_target_properties(CyRSAssistant PROPERTIES PREFIX "" SUFFIX ".addon64" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

# CyUI is not linked into the ReShade ImGui bridge; the overlay uses Conversation events.
foreach(module HTTP PLATFORM MCP AI)
  set(CYMCP_BUILD_${module} ON CACHE BOOL "" FORCE)
endforeach()
foreach(module LOCAL CHAT_UI CHAT_APP EXAMPLES TESTS BENCHMARKS)
  set(CYMCP_BUILD_${module} OFF CACHE BOOL "" FORCE)
endforeach()
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/third_party/cymcpsdk/include/cymcp/core.hpp")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/bootstrap_sdk.py" COMMAND_ERROR_IS_FATAL ANY)
endif()
add_subdirectory(third_party/cymcpsdk EXCLUDE_FROM_ALL)
set(CYRS_SDK_HEADER "${CMAKE_BINARY_DIR}/generated/sdk_catalogue.hpp")
add_custom_command(OUTPUT "${CYRS_SDK_HEADER}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/generate_sdk_registry.py" "${CYRS_SDK_HEADER}"
  DEPENDS tools/generate_sdk_registry.py companion/mcp.py tools/cymcpsdk.lock.json VERBATIM)
add_executable(CyRSAssistantCLI src/sdk_host.cpp "${CYRS_SDK_HEADER}")
target_compile_features(CyRSAssistantCLI PRIVATE cxx_std_20)
target_compile_definitions(CyRSAssistantCLI PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX CYRS_VERSION="${PROJECT_VERSION}")
target_compile_options(CyRSAssistantCLI PRIVATE /W4 /EHsc /utf-8)
target_include_directories(CyRSAssistantCLI PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(CyRSAssistantCLI PRIVATE CyMCP::Core CyMCP::Platform CyMCP::Http CyMCP::MCP CyMCP::AI windowscodecs ole32 crypt32)
target_link_options(CyRSAssistantCLI PRIVATE "/PDBALTPATH:CyRSAssistantCLI.pdb")
set_target_properties(CyRSAssistantCLI PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_custom_command(TARGET CyRSAssistantCLI POST_BUILD
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/record_sdk.py" "$<TARGET_FILE:CyRSAssistantCLI>" VERBATIM)
